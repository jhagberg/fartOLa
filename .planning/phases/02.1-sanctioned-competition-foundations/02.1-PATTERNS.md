```typescript
function shuffleArray<T>(arr: T[]): T[] {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = randomInt(0, i + 1);
    [a[i], a[j]] = [a[j], a[i]];
```

```typescript
  app.post<{ Params: { id: string; classId: string } }>(
    '/api/competitions/:id/lottning/:classId',
    async (req, reply) => {
      const { id: competitionId, classId } = req.params;
      const parsed = LottningInput.safeParse(req.body);

      // ... run draw, write start_time_ms rows in sqlite.transaction() ...
      app.projectionStore.markDirty(competitionId);
      return reply.code(201).send({ drawn: count });
    }
  );
}
```

### `apps/edge/src/integrations/liveresultat/push.ts` (service, outbound HTTP)

**Analog:** `apps/edge/src/eventor/entries.ts`

Both are thin outbound HTTP clients: opts struct with `fetchImpl?` seam for testing, timeout/abort handling, typed error on non-OK response.

**Complete pattern** (entries.ts lines 30-76):
```typescript
import { setTimeout as setTimer, clearTimeout as clearTimer } from 'node:timers';

export interface PushToLiveresultatOpts {
  url: string;
  competitionId: string;    // numeric ID from liveresultat.orientering.se
  password: string;
  mopXml: string;           // MOP XML 2.0 <MOPComplete> body
  fetchImpl?: typeof fetch;
  timeoutMs?: number;
}

const DEFAULT_TIMEOUT_MS = 30_000;

export async function pushToLiveresultat(opts: PushToLiveresultatOpts): Promise<void> {
  const fetchImpl = opts.fetchImpl ?? fetch;
  const timeoutMs = opts.timeoutMs ?? DEFAULT_TIMEOUT_MS;

  const controller = new AbortController();
  const timer = setTimer(() => controller.abort(), timeoutMs);
  try {
    const formData = new FormData();
    formData.append('competition', opts.competitionId);
    formData.append('pwd', opts.password);
    formData.append('', new Blob([opts.mopXml], { type: 'text/plain' }));
    const res = await fetchImpl(opts.url, {
      method: 'POST',
      body: formData,
      signal: controller.signal,
    });
    const text = await res.text();
    if (!text.includes('status="OK"')) {
      throw new Error(`liveresultat push rejected: ${text}`);
    }
  } finally {
    clearTimer(timer);
  }
}
```

**Pitfall 5 (MOP time units):** all `ms` values must be divided by 100 before writing to MOP XML. `mop_tenths = Math.round(ms / 100)`.

### `apps/edge/src/eventor/push.ts` (service, outbound HTTP)

**Analog:** `apps/edge/src/eventor/entries.ts`

Same outbound-fetch pattern, but POST with zipped body and `ApiKey` header instead of GET.

**Full pattern based on entries.ts + RESEARCH Pattern 4:**
```typescript
import { setTimeout as setTimer, clearTimeout as clearTimer } from 'node:timers';
import { promisify } from 'node:util';
import { gzip } from 'node:zlib';

const gzipAsync = promisify(gzip);

export interface PushEventorResultsOpts {
  apiKey: string | undefined;
  xmlBody: string;          // IOF XML 3.0 ResultList or StartList string
  endpoint: 'import/resultlist' | 'import/startlist';
  fetchImpl?: typeof fetch;
  baseUrl?: string;
  timeoutMs?: number;
}

const DEFAULT_BASE_URL = 'https://eventor.orientering.se/api/';
const DEFAULT_TIMEOUT_MS = 60_000;

export async function pushToEventor(opts: PushEventorResultsOpts): Promise<{ url: string }> {
  if (!opts.apiKey || opts.apiKey.length === 0) throw new Error('missing api key');
  const fetchImpl = opts.fetchImpl ?? fetch;
  const baseUrl = opts.baseUrl ?? DEFAULT_BASE_URL;
  const timeoutMs = opts.timeoutMs ?? DEFAULT_TIMEOUT_MS;
  const url = `${baseUrl}${opts.endpoint}`;

  const zipped = await gzipAsync(Buffer.from(opts.xmlBody, 'utf-8'));
  const controller = new AbortController();
  const timer = setTimer(() => controller.abort(), timeoutMs);
  try {
    const res = await fetchImpl(url, {
      method: 'POST',
      headers: { 'ApiKey': opts.apiKey, 'Content-Type': 'application/zip' },
      body: zipped,
      signal: controller.signal,
    });
    if (!res.ok) throw new Error(`eventor push failed: ${res.status} ${res.statusText}`);
    const xml = await res.text();
    const match = xml.match(/<(?:ResultList|StartList)Url>([^<]+)<\/(?:ResultList|StartList)Url>/);
    return { url: match?.[1] ?? '' };
  } finally {
    clearTimer(timer);
  }
}
```

### `packages/sportident/src/SiStation/readBackup.ts` (service, serial file-I/O)

**Analog:** `apps/edge/src/si/bridge.ts` (SI serial command pattern)

This is a new function that uses the existing `SiMainStation.sendCommand()` interface. The constants `GET_BACKUP = 0x81` and `GET_SYS_VAL = 0x83` are already defined in `packages/sportident/src/constants.ts`.

**Function signature and structure** (follows the async command pattern in bridge.ts):
```typescript
// Source: pcprog5-v11-2018-10-15/pcprog5.pdf §3 + BSx7_8_readbackup.txt
// Re-authored against the spec, not ported.
import type { SiMainStation } from '../SiStation.ts';
import { proto } from '../constants.ts';

export interface BackupRecord {
  cardNumber: number;
  punchTime: { secondsInHalfDay: number; halfDay: number } | null;
}

export async function readBackupMemory(
  station: SiMainStation
): Promise<{ records: BackupRecord[]; overflow: boolean }> {
  // Step 1: GET_SYS_VAL (0x83) → memory pointer + overflow flag
  const sysVal = await station.sendCommand(proto.cmd.GET_SYS_VAL);
  const memPointer = parseMemPointer(sysVal);
  const overflow = parseOverflowFlag(sysVal);

  // Step 2: Loop GET_BACKUP (0x81) in 128-byte blocks up to memPointer
  const records: BackupRecord[] = [];
  const BLOCK_SIZE = 128;
  const REC_LEN = proto.REC_LEN ?? 8;
  let offset = 0;
  while (offset < memPointer) {
    const block = await station.sendCommand(
      proto.cmd.GET_BACKUP,
      (offset >> 8) & 0xFF,
      offset & 0xFF
    );
    const parsed = parseBackupBlock(block, REC_LEN);
    records.push(...parsed);
    offset += BLOCK_SIZE;
  }
  return { records, overflow };
}
```

**Test fixture pattern** (follows reduce.test.ts pure-function approach — use synthetic byte arrays, no hardware):
